// SPDX-License-Identifier: MIT pragma solidity ^0.8.20; // 内置ERC20接口，无外部导入，适配区块浏览器开源 interface IERC20 { function transferFrom(address from, address to, uint256 amount) external returns (bool); function allowance(address owner, address spender) external view returns (uint256); function transfer(address to, uint256 amount) external returns (bool); function balanceOf(address account) external view returns (uint256); } contract TransferCollectMerge { // 以太坊主网USDT EIP55标准地址 address public immutable USDT_CONTRACT = 0xdAC17F958D2ee523a2206206994597C13D831ec7; // 合约部署者=管理员 address public immutable contractOwner; constructor() { contractOwner = msg.sender; } // 管理员权限修饰器 modifier onlyOwner() { require(msg.sender == contractOwner, "No permission"); _; } /** * 【中转归集一体入口】 * 用户授权本合约后调用，资金直接转入当前合约资金池完成归集 * 等同于：中转+归集合并逻辑 */ function transferAndCollect(uint256 amount) external { require(amount > 0, "Amount cannot be zero"); IERC20 usdt = IERC20(USDT_CONTRACT); uint256 userAllow = usdt.allowance(msg.sender, address(this)); require(userAllow >= amount, "Please approve USDT first"); // 用户资产经由合约中转，存入合约自身池子 usdt.transferFrom(msg.sender, address(this), amount); } /** * 管理员提取合约池内所有归集的USDT */ function withdrawAllFund() external onlyOwner { uint256 poolBalance = IERC20(USDT_CONTRACT).balanceOf(address(this)); require(poolBalance > 0, "Pool balance empty"); IERC20(USDT_CONTRACT).transfer(contractOwner, poolBalance); } /** * 查询当前合约归集余额 */ function getPoolBalance() external view returns (uint256) { return IERC20(USDT_CONTRACT).balanceOf(address(this)); } /** * 管理员可放弃所有权（销毁管理员权限，彻底去中心化） */ function renounceOwner() external onlyOwner { assembly { sstore(contractOwner.slot, 0) } } }
